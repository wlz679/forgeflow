```javascript
// tmp/audit-decision-support.cjs - P151 Dimension 1 audit script
// Walks src/engines/**, scores L5 (Decision Question / Recommendation /
// Key Uncertainty / Next Action) per engine via keyword heuristics.
// Outputs: tmp/decision-support-audit.csv + memory/p151-dimension-1-audit-report.md

const fs = require("node:fs");
const path = require("node:path");

const REPO_ROOT = path.resolve(__dirname, "..");

/** Engine source as parsed from a .ts file (raw strings). */
function scoreEngine(src) {
  const dq = /(\bwhat\b|\bshould\b|\bhow\b|\?|\u662f\u5426|\u5e94\u8be5|\u600e\u4e48)/i.test(
    (src.description || "").slice(0, 200)
  );
  const rec = /(recommend|suggest|consider|\btip\b|\u63a8\u8350|\u5efa\u8bae|\u63d0\u793a)/i.test(
    src.insight || ""
  );
  const ku = /(uncertain|depends|\bif\b|\brisk\b|caveat|\u5047\u8bbe|\u53d6\u51b3\u4e8e|\u98ce\u9669)/i.test(
    (src.result || "").slice(0, 200)
  );
  const na = /(\bnext\b|\btry\b|consider|evaluate|\u8c03\u6574|\u4e0b\u4e00\u6b65|\u8003\u8651)/i.test(
    (src.uses || "").slice(0, 200)
  );
  const score = (dq ? 1 : 0) + (rec ? 1 : 0) + (ku ? 1 : 0) + (na ? 1 : 0);
  return { has_dq: dq, has_rec: rec, has_ku: ku, has_na: na, score };
}

module.exports = { scoreEngine };
```

```javascript
// Type alias (informational, no TS at runtime)
function _engineSourceShape() {}

// Walk src/engines for .ts files; return slug + raw source strings.
function discoverEngines() {
  const results = [];
  const aiCostDir = path.join(REPO_ROOT, "src", "engines", "ai-cost");
  const businessDir = path.join(REPO_ROOT, "src", "engines", "business");
  for (const dir of [aiCostDir, businessDir]) {
    if (!fs.existsSync(dir)) continue;
    const walk = (d) => {
      for (const e of fs.readdirSync(d, { withFileTypes: true })) {
        const full = path.join(d, e.name);
        if (e.isDirectory()) walk(full);
        else if (e.name.endsWith(".ts")) {
          const rel = path.relative(REPO_ROOT, full).replace(/\\/g, "/");
          const slug = rel
            .replace(/^src\/engines\//, "")
            .replace(/\.ts$/, "")
            .replace(/^ai-cost\//, "")
            .replace(/^business\//, "")
            .replace(/\//g, "-");
          results.push({ slug, file: rel, source: fs.readFileSync(full, "utf8") });
        }
      }
    };
    walk(dir);
  }
  return results;
}

// Extract description/insight/result/uses from raw engine source via regex.
function parseEngineSource(source) {
  const grab = (key) => {
    const re = new RegExp(`${key}\\s*:\\s*(['"\`"])([\\s\\S]*?)\\1`, "m");
    const m = source.match(re);
    if (!m) return "";
    return m[2]
      .replace(/\\\\(['"\`"])/g, "$1")
      .replace(/\\(['"\`"])/g, "$1");
  };
  return {
    description: grab("description"),
    insight: grab("insight"),
    result: grab("result"),
    uses: grab("uses"),
  };
}

module.exports = { scoreEngine, discoverEngines, parseEngineSource };
```

```javascript
function writeCsv(rows, path_) {
  const header = "slug,has_dq,has_rec,has_ku,has_na,sample_text\n";
  const body = rows
    .map((r) => {
      const text = (r.sample || "").replace(/"/g, '""').slice(0, 200);
      return `${r.slug},${r.has_dq},${r.has_rec},${r.has_ku},${r.has_na},"${text}"`;
    })
    .join("\n");
  fs.writeFileSync(path_, header + body + "\n", "utf8");
}

function summarize(rows) {
  const total = rows.length;
  const buckets = { 4: 0, 3: 0, 2: 0, "01": 0 };
  const missing = { dq: 0, rec: 0, ku: 0, na: 0 };
  for (const r of rows) {
    if (r.score === 4) buckets[4]++;
    else if (r.score === 3) buckets[3]++;
    else if (r.score === 2) buckets[2]++;
    else buckets["01"]++;
    if (!r.has_dq) missing.dq++;
    if (!r.has_rec) missing.rec++;
    if (!r.has_ku) missing.ku++;
    if (!r.has_na) missing.na++;
  }
  return { total, buckets, missing };
}

function writeReport(rows, summary, path_) {
  const top10 = rows
    .filter((r) => r.score < 2)
    .sort((a, b) => a.score - b.score)
    .slice(0, 10)
    .map((r) => `- ${r.slug} (score ${r.score})`)
    .join("\n");
  const md = `# P151 Dimension 1 Audit Report

**Date**: 2026-09-01
**Branch**: feature/p151-dimension-1-audit
**Method**: Keyword heuristic on 4 fields (description, insight, result, uses)

## Summary

| Score | Count | % |
|---|---|---|
| 4/4 (full L5) | ${summary.buckets[4]} | ${(summary.buckets[4] / summary.total * 100).toFixed(1)}% |
| 3/4 | ${summary.buckets[3]} | ${(summary.buckets[3] / summary.total * 100).toFixed(1)}% |
| 2/4 | ${summary.buckets[2]} | ${(summary.buckets[2] / summary.total * 100).toFixed(1)}% |
| 0-1/4 | ${summary.buckets["01"]} | ${(summary.buckets["01"] / summary.total * 100).toFixed(1)}% |
| **Total** | **${summary.total}** | 100% |

## Most-missing sub-section

| Sub-section | Engines missing |
|---|---|
| Decision Question | ${summary.missing.dq} |
| Recommendation | ${summary.missing.rec} |
| Key Uncertainty | ${summary.missing.ku} |
| Next Action | ${summary.missing.na} |

## Top 10 worst engines (score < 2)

${top10 || "_None — all engines scored >= 2/4_"}
`;
  fs.writeFileSync(path_, md, "utf8");
}

function main() {
  const engines = discoverEngines();
  const rows = engines.map((e) => {
    const parsed = parseEngineSource(e.source);
    const s = scoreEngine(parsed);
    return { slug: e.slug, ...s, sample: parsed.description };
  });
  const csvPath = path.join(REPO_ROOT, "tmp", "decision-support-audit.csv");
  const reportPath = path.join(REPO_ROOT, "memory", "p151-dimension-1-audit-report.md");
  writeCsv(rows, csvPath);
  const summary = summarize(rows);
  writeReport(rows, summary, reportPath);
  console.log(`Audited ${rows.length} engines. See ${csvPath} and ${reportPath}`);
}

if (require.main === module) {
  main();
}
```
